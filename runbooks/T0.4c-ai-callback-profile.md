# Runbook T0.4c — Which engine callbacks make the AI worker slow (worker t02)

Read `runbooks/_wave4-common.md` first. Report: `reports/T0.4c.md`. Logs: `runs/T0.4c*`.
Files you may edit: `tools/wasm/ai/ai-worker.js` ONLY, and only as Step 1 says.

Background: with `ZEROAD_AI_REMOTE=1` the AI runs in a V8 worker. 68% of its time is spent in
synchronous callbacks to the engine (1.07 M calls). We need the callbacks broken down by name so the
supervisor can design caching/batching. The new counters must only run when `ZEROAD_AI_WORKER_STATS` is set.

## Step 1 — Add per-callback counters to `tools/wasm/ai/ai-worker.js`
1a. Find this block (around line 86):
```js
const stats = { calls: 0, ms: 0, cached: 0 };
let writeStats = null;
if (isNode && process.env.ZEROAD_AI_WORKER_STATS) {
	const fs = require('node:fs');
	const fd = fs.openSync(process.env.ZEROAD_AI_WORKER_STATS, 'w');
	writeStats = (line) => fs.writeSync(fd, line + '\n');
}
```
Replace it with:
```js
const stats = { calls: 0, ms: 0, cached: 0 };
let writeStats = null;
// Per-callback totals (same switch): <file>.keys.json, rewritten every 200 requests.
const keyStats = new Map();
let dumpKeys = null;
let requestCount = 0;
if (isNode && process.env.ZEROAD_AI_WORKER_STATS) {
	const fs = require('node:fs');
	const fd = fs.openSync(process.env.ZEROAD_AI_WORKER_STATS, 'w');
	writeStats = (line) => fs.writeSync(fd, line + '\n');
	dumpKeys = () => fs.writeFileSync(process.env.ZEROAD_AI_WORKER_STATS + '.keys.json',
		JSON.stringify(Object.fromEntries([...keyStats].sort((a, b) => b[1].ms - a[1].ms)), null, 1));
}
function rpcKey(request) {
	switch (request.op) {
	case 'engine': return `engine:${request.name}`;
	case 'simcall':
	case 'simget': return `${request.op}:${request.root}.${request.path.join('.')}`;
	case 'qi': return `qi:${request.iid}`;
	case 'qicall': return `qicall:${request.iid}.${request.method}`;
	default: return String(request.op);
	}
}
function countKey(request, ms, cached) {
	const key = rpcKey(request);
	let entry = keyStats.get(key);
	if (!entry)
		keyStats.set(key, entry = { calls: 0, ms: 0, cached: 0 });
	if (cached)
		++entry.cached;
	else {
		++entry.calls;
		entry.ms += ms;
	}
}
```
1b. Replace the function `rpc` with:
```js
function rpc(request) {
	const t0 = writeStats ? performance.now() : 0;
	const response = Codec.decode(callEngine(SERVICE_DISPATCH, Codec.encode(request)));
	if (writeStats)
		countKey(request, performance.now() - t0, false);
	if (response.error !== undefined)
		throw new Error(response.error);
	return response.v;
}
```
1c. In `rpcCached`, change the cache-hit branch to:
```js
	if (turnCache.has(key)) {
		++stats.cached;
		if (writeStats)
			countKey(request, 0, true);
		return turnCache.get(key);
	}
```
1d. In `serve()`, directly after the two lines `if (writeStats)` / `writeStats(...)`, add:
```js
		if (dumpKeys && ++requestCount % 200 === 0)
			dumpKeys();
```
1e. Check syntax: `node --check tools/wasm/ai/ai-worker.js && echo SYNTAX-OK`.

## Step 4 — Tables
