## Step 2 — Sizes by folder, by extension, and by civilization

```bash
cd ~/DEVELOPMENT/0ad-web/upstream/binaries/data/mods
python3 -I - public/public.zip mod/mod.zip <<'PY' > ~/DEVELOPMENT/0ad-web/runs/T4.3a-sizes.txt
import sys, zipfile, collections
civs = ['athen','brit','cart','gaul','iber','kush','mace','maur','pers','ptol','rome','sele','spart','han']
for path in sys.argv[1:]:
    z = zipfile.ZipFile(path)
    by_dir2, by_ext, by_civ = collections.Counter(), collections.Counter(), collections.Counter()
    total = 0
    for i in z.infolist():
        if i.is_dir():
            continue
        total += i.file_size
        parts = i.filename.split('/')
        by_dir2['/'.join(parts[:2]) if len(parts) > 2 else parts[0]] += i.file_size
        by_ext[i.filename.rsplit('.', 1)[-1].lower() if '.' in parts[-1] else '(none)'] += i.file_size
        low = i.filename.lower()
        hits = [c for c in civs if f'/{c}/' in low or f'/{c}_' in low or f'_{c}_' in low or f'{c}.' in parts[-1].lower()]
        by_civ[hits[0] if len(hits) == 1 else ('(shared)' if not hits else '(several)')] += i.file_size
    mb = lambda b: f'{b/1e6:10.1f} MB'
    print(f'##### {path}: {len(z.infolist())} entries, {mb(total)}')
    print('--- top 40 folders (2 levels)')
    for k, v in by_dir2.most_common(40): print(mb(v), k)
    print('--- extensions')
    for k, v in by_ext.most_common(25): print(mb(v), k)
    print('--- civilizations (by file name/path match)')
    for k, v in by_civ.most_common(): print(mb(v), k)
PY
echo "EXIT=$?"; wc -l ~/DEVELOPMENT/0ad-web/runs/T4.3a-sizes.txt
```

## Step 3 — Maps

```bash
cd ~/DEVELOPMENT/0ad-web/upstream/binaries/data/mods
python3 -I -c "
import zipfile, collections
z = zipfile.ZipFile('public/public.zip'); c = collections.Counter()
for i in z.infolist():
    if i.filename.startswith('maps/'): c[i.filename.rsplit('.',1)[0]] += i.file_size
print(len(c), 'map entries'); [print(f'{v/1e6:8.2f} MB', k) for k, v in c.most_common(15)]
print('acropolis:', [(k, v) for k, v in c.items() if 'acropolis' in k])
" > ~/DEVELOPMENT/0ad-web/runs/T4.3a-maps.txt; echo "EXIT=$?"
```
